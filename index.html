// KeyValidator.cpp — xác thực key qua Firebase Realtime Database
// Build: g++ KeyValidator.cpp -o keyval -lcurl -ljsoncpp

#include <curl/curl.h>
#include <json/json.h>
#include <string>
#include <ctime>
#include <iostream>

static size_t WriteCb(void* c, size_t s, size_t n, std::string* out) {
    out->append((char*)c, s * n);
    return s * n;
}

struct KeyInfo {
    bool     active;
    std::string pkg;
    long     expiry;   // unix timestamp
};

KeyInfo ValidateKey(const std::string& key) {
    CURL* h = curl_easy_init();
    if (!h) return {false, "", 0};

    std::string url = "https://keyb-2f31d-default-rtdb.asia-southeast1"
                     ".firebasedatabase.app/keys/" + key + ".json";
    std::string body;

    curl_easy_setopt(h, CURLOPT_URL, url.c_str());
    curl_easy_setopt(h, CURLOPT_WRITEFUNCTION, WriteCb);
    curl_easy_setopt(h, CURLOPT_WRITEDATA, &body);
    curl_easy_setopt(h, CURLOPT_TIMEOUT, 5L);

    CURLcode rc = curl_easy_perform(h);
    long code = 0;
    curl_easy_getinfo(h, CURLINFO_RESPONSE_CODE, &code);
    curl_easy_cleanup(h);

    if (rc != CURLE_OK || code != 200) return {false, "", 0};

    Json::Value root;
    Json::CharReaderBuilder rb;
    std::string err;
    std::istringstream is(body);
    if (!Json::parseFromStream(rb, is, &root, &err)) return {false, "", 0};

    KeyInfo info{};
    info.active = root.get("status", "inactive").asString() == "active";
    info.pkg    = root.get("package", "").asString();
    info.expiry = root.get("expiry", 0).asInt64();

    // kiểm tra hết hạn
    if (info.active && info.expiry < std::time(nullptr)) {
        info.active = false;
    }
    return info;
}

// KeyValidator.java — Android native
// Chạy trong AsyncTask / Coroutine để tránh NetworkOnMainThreadException

import java.net.HttpURLConnection;
import java.net.URL;
import java.io.BufferedReader;
import java.io.InputStreamReader;
import org.json.JSONObject;

public class KeyValidator {
    private static final String BASE = "https://keyb-2f31d-default-rtdb.asia-southeast1"
                                       + ".firebasedatabase.app/keys/";

    public static class Result {
        public boolean active;
        public String pkg;
        public long expiry;
    }

    public static Result validate(String key) throws Exception {
        Result r = new Result();
        HttpURLConnection c = (HttpURLConnection) new URL(BASE + key + ".json").openConnection();
        c.setConnectTimeout(5000);
        c.setReadTimeout(5000);

        if (c.getResponseCode() != 200) return r;

        StringBuilder sb = new StringBuilder();
        try (BufferedReader br = new BufferedReader(new InputStreamReader(c.getInputStream()))) {
            String line;
            while ((line = br.readLine()) != null) sb.append(line);
        }
        JSONObject o = new JSONObject(sb.toString());
        r.active  = "active".equals(o.optString("status"));
        r.pkg     = o.optString("package");
        r.expiry  = o.optLong("expiry", 0L);
        if (r.active && r.expiry < System.currentTimeMillis() / 1000) r.active = false;
        return r;
    }
}

# key_validator.py — Python (requests)
# pip install requests

import time
import requests

BASE = "https://keyb-2f31d-default-rtdb.asia-southeast1"
FIREBASE = f"{BASE}.firebasedatabase.app/keys/"

def validate_key(key: str) -> dict:
    url = f"{FIREBASE}{key}.json"
    try:
        resp = requests.get(url, timeout=5)
        if resp.status_code != 200:
            return {"active": False}
        data = resp.json() or {}
    except requests.RequestException:
        return {"active": False}

    active  = data.get("status") == "active"
    pkg     = data.get("package", "")
    expiry  = int(data.get("expiry", 0))
    if active and expiry < time.time():
        active = False

    return {"active": active, "package": pkg, "expiry": expiry}

// KeyGui.cpp — ImGui overlay menu cho Key Manager
// Build với Dear ImGui + GLFW/OpenGL3 hoặc DirectX

#include <imgui.h>
#include <string>
#include <ctime>
#include <cstring>
#include "KeyValidator.cpp"  // ValidateKey() ở trên

struct Session {
    KeyInfo info{};
    bool    checked = false;
    char    keyBuf[64] = "";
};

static void RenderCountdown(long expiry) {
    long now = std::time(nullptr);
    long left = expiry - now;
    if (left <= 0) {
        ImGui::TextColored(ImVec4(1, 0.4f, 0.4f, 1), "● ĐÃ HẾT HẠN");
        return;
    }
    int d = left / 86400;
    int h = (left % 86400) / 3600;
    int m = (left % 3600) / 60;

    ImGui::Text("Còn lại:");
    ImGui::SameLine();
    ImGui::TextColored(ImVec4(0.2f, 1, 0.4f, 1),
                       "%d ngày %02d:%02d", d, h, m);
}

void DrawKeyMenu(Session& s) {
    ImGui::SetNextWindowSize(ImVec2(360, 280), ImGuiCond_FirstUseEver);
    ImGui::Begin("Key Manager Pro  v2.0",
                 nullptr,
                 ImGuiWindowFlags_NoCollapse);

    // ----- HEADER -----
    ImGui::TextColored(ImVec4(0.4f, 0.7f, 1, 1), "◆ KEY MANAGER");
    ImGui::Separator();

    // ----- STATUS -----
    if (s.info.active) {
        ImGui::TextColored(ImVec4(0.2f, 1, 0.4f, 1), "● ĐÃ KÍCH HOẠT");
        ImGui::Text("Gói:    %s", s.info.pkg.c_str());

        ImGui::Spacing();
        RenderCountdown(s.info.expiry);

        // Thanh tiến trình
        long total = s.info.expiry - (std::time(nullptr) - 30L * 86400);
        long used  = std::time(nullptr) - (s.info.expiry - total);
        float frac  = total > 0 ? (float)used / total : 0;
        ImGui::ProgressBar(frac, ImVec2(-1, 0), "");
    }
    else {
        ImGui::TextColored(ImVec4(1, 0.6f, 0.2f, 1), "● CHƯA KÍCH HOẠT");
        ImGui::Spacing();

        ImGui::InputText("##keyinput", s.keyBuf, sizeof(s.keyBuf));
        ImGui::SameLine();
        if (ImGui::Button("Kích hoạt")) {
            s.info    = ValidateKey(s.keyBuf);
            s.checked = true;
        }

        if (s.checked && !s.info.active) {
            ImGui::TextColored(ImVec4(1, 0.4f, 0.4f, 1),
                               "Key không hợp lệ hoặc đã hết hạn");
        }
    }

    ImGui::Spacing();
    ImGui::Separator();
    ImGui::TextDisabled("UID: a1b2c3d4e5  ·  v2.0.0");
    ImGui::End();
}

// Gọi DrawKeyMenu(session) mỗi frame trong vòng lặp render ImGui.
